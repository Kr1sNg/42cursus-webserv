Manual curl examples

# GET
curl -v "http://localhost:8080/echo?hello=world"

# POST (form data)
curl -v -X POST -F "name=tat" -F "message=hi" http://localhost:8080/submit

# POST (JSON)
curl -v -X POST -H "Content-Type: application/json" -d '{"color":"blue"}' http://localhost:8080/api/json

# DELETE
curl -v -X DELETE http://localhost:8080/resource/123
